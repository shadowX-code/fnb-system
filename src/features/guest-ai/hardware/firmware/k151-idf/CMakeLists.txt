cmake_minimum_required(VERSION 3.16)
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
set(FEEDX_BUILD_PROFILE "normal" CACHE STRING "normal or fault-test")
if(NOT FEEDX_BUILD_PROFILE STREQUAL "normal" AND NOT FEEDX_BUILD_PROFILE STREQUAL "fault-test")
  message(FATAL_ERROR "FEEDX_BUILD_PROFILE must be normal or fault-test")
endif()
project(feedx_k151_guest_ai_bridge)
if(FEEDX_BUILD_PROFILE STREQUAL "fault-test" AND NOT CONFIG_FEEDX_GUEST_AI_FAULT_INJECTION)
  message(FATAL_ERROR "fault-test profile requires CONFIG_FEEDX_GUEST_AI_FAULT_INJECTION=y")
endif()
if(FEEDX_BUILD_PROFILE STREQUAL "normal" AND CONFIG_FEEDX_GUEST_AI_FAULT_INJECTION)
  message(FATAL_ERROR "normal profile requires CONFIG_FEEDX_GUEST_AI_FAULT_INJECTION=n")
endif()
